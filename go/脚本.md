





## 查看生产环境上哪些go程序开启了DEBUG

```bash
#!/bin/bash
# 一键扫描所有非内核进程，查找开启 GODEBUG 的进程

echo "正在扫描所有用户进程..."
for pid in $(ls /proc | grep -E '^[0-9]+$'); do
    # 跳过无法访问的进程 (权限不足或已退出)
    if [ ! -r "/proc/$pid/environ" ]; then
        continue
    fi

    # 快速检查: 如果 environ 文件中完全没提到 GODEBUG，直接跳过 (提高效率)
    if ! grep -q "GODEBUG" "/proc/$pid/environ" 2>/dev/null; then
        continue
    fi

    # 获取进程命令名称 (用于显示)
    cmd=$(cat /proc/$pid/cmdline 2>/dev/null | tr '\0' ' ' | head -c 50)
    
    # 如果是内核线程 (cmdline 通常为空或很短)，跳过
    if [ -z "$cmd" ]; then
        continue
    fi

    echo "------------------------------------------"
    echo "🎯 发现目标 PID: $pid"
    echo "📝 进程命令: $cmd"
    echo "🔍 GODEBUG 设置内容:"
    cat /proc/$pid/environ 2>/dev/null | tr '\0' '\n' | grep GODEBUG
done

echo "------------------------------------------"
echo "扫描完成。如果没有输出以上内容，说明当前没有进程设置 GODEBUG。"
```
